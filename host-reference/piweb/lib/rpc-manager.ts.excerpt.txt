  private notifyAgentRunCompleteIfIdle(): void {
    if (!this.agentRunNeedsCompletion || this.isRunning()) return;
    this.agentRunNeedsCompletion = false;
    const suggest = this.suggestionAfterSettle;
    this.suggestionAfterSettle = false;
    // A stopped run did not finish a task, as pi's own status reporting has it: nothing to announce.
    if (this.suppressCompletionNotifications || this.lastRunAborted) return;
    if (suggest) void this.startPromptSuggestion().catch((error: unknown) => {
      console.error("[pi-web] prompt suggestion settings failed:", error instanceof Error ? error.message : error);
    });
    // A subagent of this session is still at work: each report starts a run here, and the run
    // after the last one is the end of the task (#1434).
    if (hasUnsettledSubagentRun(this.sessionId)) return;
    try {
      this.onAgentRunComplete?.(this.sessionId, this.inner.sessionManager.getSessionName());
    } catch (error) {
      console.error("[pi-web] completion listener failed:", error instanceof Error ? error.message : error);
    }
  }

  private async startPromptSuggestion(): Promise<void> {
    const inner = this.inner;
    const selectedModel = inner.model;
    const messages = inner.messages;
    const complete = inner.modelRuntime?.completeSimple;
    const request = this.suggestionRequest;
    if (!selectedModel || !messages?.length || !complete || !request || inner.pendingMessageCount > 0) return;
    const last = messages.at(-1);
    if (last?.role !== "assistant" || last.stopReason !== "stop" ||
        !last.content.some((block) => block.type === "text" && block.text.trim())) return;
    const leaf = inner.sessionManager.getLeafId();
    const revision = this.suggestionSettingsRevision;
    const isCurrent = () => this.isAlive() && !this.isRunning() && inner.pendingMessageCount === 0 && inner.model === selectedModel &&
      this.suggestionSettingsRevision === revision && this.suggestionRequest === request &&
      inner.sessionManager.getLeafId() === leaf && inner.messages?.at(-1) === last;
    const settings = await readPromptSuggestionSettings();
    if (!settings.enabled || !isCurrent()) return;
    const [api, models] = await Promise.all([
      loadPromptSuggestionPlugin(),
      sessionSuggestionModels(inner.modelRuntime as ModelRuntime, inner.settingsManager.getEnabledModels()),
    ]);
    if (!isCurrent()) return;
    api.validatePredictionSelection(settings, models.map((item) => item.option));
    const model = models.find((item) => item.option.provider === settings.provider && item.option.id === settings.modelId)!.model;
    const compacted = api.buildCompactedSuggestionContext(inner.sessionManager.getBranch(), inner.sessionManager.buildSessionProjection().entries);
    this.promptSuggestions.start({
      enabled: true,
      model,
      thinkingLevel: settings.thinkingLevel,
      preferences: settings.preferences,
      messages: [...request.messages, last],
      compaction: compacted && { id: compacted.id, messages: convertToLlm(compacted.messages) },
      sessionId: this.sessionId,
      agentDir: getAgentDir(),
      revision: leaf ?? String(last.timestamp),
      complete: (targetModel, context, options) => complete.call(inner.modelRuntime, targetModel, context, options),
      isCurrent,
    });
  }

  onEvent(listener: EventListener): () => void {
    this.listeners.add(listener);

    return () => {
      this.listeners.delete(listener);
